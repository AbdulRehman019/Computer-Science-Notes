select distinct(year(birth_date)) AS birth from patients order by birth;

select first_name, last_name, birth_date from patients where year(birth_date) >= 1970 AND year(birth_date) < 1980 order by birth_date;

select first_name, last_name, height from patients where height = (select max(height) from patients);

select patient_id, count(patient_id) from admissions where patient_id = 579 group by patient_id;

SELECT first_name, last_name, COUNT(*) AS duplicate_count
FROM patients
GROUP BY first_name, last_name
HAVING COUNT(*) > 1;


SELECT 
    ROUND(
        (SUM(CASE WHEN gender = 'M' THEN 1 ELSE 0 END) * 100.0) / COUNT(*),
        2
    ) AS male_percentage
FROM patients;


select admission_date, 
admission_count, 
admission_count - LAG(admission_count) OVER (ORDER BY admission_date) AS change_from_previous_day 
from (select admission_date, count(*) as admission_count from admissions group by admission_date) true
order by admission_date;
