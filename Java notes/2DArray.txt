
package com.Abdul;

import java.util.Arrays;
import java.util.Scanner;

public class MultiDimensionArray {
    public static void main(String[] args) {

        Scanner in = new Scanner(System.in);


//        int[][] arr = {
//                {1,2,3},   // 0th index
//                {4,5},     // 1st index
//                {6,7,8,9}  // 2nd index    arr[2] = {6,7,8,9}    arr[2][0] = 6;
//        };



        int[][] arr = new int[3][2];

        // input

        for(int row = 0; row < arr.length; row++) {     // arr.length will give number of rows
            for(int col = 0; col < arr[row].length; col++) {
                arr[row][col] = in.nextInt();
            }
        }

        //Output

        for(int row = 0; row < arr.length; row++) {
            for(int col = 0; col < arr[row].length; col++) {
                System.out.print(arr[row][col] + " ");
            }
            System.out.println();
        }


        for(int row = 0; row < arr.length; row++) {
            System.out.println(Arrays.toString(arr[row]));
        }

        for(int[] a: arr) {
            System.out.println(Arrays.toString(a));
        }
    }
}



/*
    2D array can be considered as an array of array.
    Here arr will be stored in stack as in 1D array.
    Reference arr will be stored in heap which in turn will point to the actual array at different position in heap.
    (Similar to 1D array [Refer to Array.txt]).
    arr[0], arr[1], ... , arr[n] will be stored in heap and, it will point to the actual arrays. For example:
    arr[0] will point to {1,2,3} and arr[0][0] will point to 1.

 */
